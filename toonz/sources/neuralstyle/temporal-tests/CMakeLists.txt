cmake_minimum_required(VERSION 3.18)
project(NeuralTemporalQualification LANGUAGES CXX)
find_package(Qt5 REQUIRED COMPONENTS Core Gui)
find_path(OPENCV_HEADERS opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
find_library(OPENCV_CORE opencv_core REQUIRED)
find_library(OPENCV_IMGPROC opencv_imgproc REQUIRED)
find_library(OPENCV_VIDEO opencv_video REQUIRED)
add_executable(neural-temporal-tests main.cpp ../../stdfx/neuraltemporal.cpp)
target_compile_features(neural-temporal-tests PRIVATE cxx_std_17)
target_include_directories(neural-temporal-tests PRIVATE ${OPENCV_HEADERS})
target_link_libraries(neural-temporal-tests PRIVATE Qt5::Core Qt5::Gui ${OPENCV_CORE} ${OPENCV_IMGPROC} ${OPENCV_VIDEO})
